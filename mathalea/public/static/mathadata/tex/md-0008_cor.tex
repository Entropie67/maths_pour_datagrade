			\begin{MADenumerate}

				\item Vérification de l'appartenance des points à la droite $d : y = \frac{2}{3}x - \frac{7}{3}$

				\begin{itemize}
					\item $A(5;1)$ : $y = \frac{2}{3}\cdot 5 - \frac{7}{3} = 1$ → appartient à la droite
					\item $B(3;2)$ : $y = \frac{2}{3}\cdot 3 - \frac{7}{3} = -\frac{1}{3}$ → n'appartient pas
					\item $C(5;-3)$ : $y = 1 \neq -3$ → n'appartient pas
					\item $D(-1;-3)$ : $y = \frac{2}{3}\cdot (-1) - \frac{7}{3} = -3$ → appartient à la droite
					\item $E(0,5;-2)$ : $y = \frac{2}{3}\cdot 0,5 - \frac{7}{3} = -2$ → appartient à la droite
					\item $F(-1;-2)$ : $y = \frac{2}{3}\cdot(-1) - \frac{7}{3} = -3 \neq -2$ → n'appartient pas
				\end{itemize}

				\item Couleur des points :
				\begin{itemize}
					\item Vert : $B, F$
					\item Bleu : $A, C, D, E$
				\end{itemize}

				\item Bonus :
				\textbf{ Point $G$ à coordonnées entières :} On cherche $G$ avec $x_G$ entier tel que $2 x_G - 7$ est divisible par 3. On trouve $x_G = 2$ et $y_G = -1$.

				Donc $G(2;-1)$ → Bleu.

				\textbf{Point $H$ d'ordonnée comprise entre -0,5 et 0,5 :} On choisit $y_H = 0$, alors on résout pour $x_H$.
				$$
					0 = \frac{2}{3}x_H - \frac{7}{3} \implies x_H = \frac{7}{2} = 3,5
				$$

				Donc $H(3,5;0)$→ Bleu.

			\end{MADenumerate}
