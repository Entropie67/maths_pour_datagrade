			\begin{MADenumerate}
				\item \textbf{Calcul de l'équation de la droite $(AB)$ :}

				\begin{enumerate}
					\item \textbf{Coefficient directeur $m$ :}

					      On utilise la formule $m = \dfrac{y_B - y_A}{x_B - x_A}$ avec $A(0;1)$ et $B(-2;-3)$.

					      $$m = \dfrac{-3 - 1}{-2 - 0} = \dfrac{-4}{-2} = 2$$

					\item \textbf{Ordonnée à l'origine $p$ :}

					      On remplace les coordonnées de $A(0;1)$ dans $y = mx + p$ :

					      $$1 = 2 \times 0 + p$$
					      $$1 = p$$

					      Donc $p = 1$.

					\item \textbf{Équation réduite :}

					      $$\boxed{(AB) : y = 2x + 1}$$
				\end{enumerate}

				\item \textbf{Position des points $C$, $D$ et $E$ par rapport à la droite $(AB)$ :}

				\begin{enumerate}
					\item \textbf{Vérification pour chaque point :}

					      \textbf{Point $C(-3;2)$ :}

					      Pour $x = -3$ : $y = 2 \times (-3) + 1 = -6 + 1 = -5$

					      Comme $2 \neq -5$, le point $C$ \textbf{n'est pas aligné} avec $A$ et $B$.

					      \textbf{Point $D(1;-4)$ :}

					      Pour $x = 1$ : $y = 2 \times 1 + 1 = 2 + 1 = 3$

					      Comme $-4 \neq 3$, le point $D$ \textbf{n'est pas aligné} avec $A$ et $B$.

					      \textbf{Point $E(1;3)$ :}

					      Pour $x = 1$ : $y = 2 \times 1 + 1 = 2 + 1 = 3$

					      Comme $3 = 3$, le point $E$ \textbf{est aligné} avec $A$ et $B$.

					\item \textbf{Conclusion :}

					      Seul le point $E$ est aligné avec $A$ et $B$.
				\end{enumerate}

				\item \textbf{Voir graphique.}

				\item $C$ est vert, $A$, $B$, $D$ et $E$ sont bleus.

			\end{MADenumerate}
